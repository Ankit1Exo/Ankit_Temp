DROP TABLE IF EXISTS #src;

INTO #src

    SELECT s0.*,
        STUFF((SELECT '' + SUBSTRING(s0.[Social_Security_Number], v.number, 1)
               FROM master..spt_values v
               WHERE v.type = 'P' AND v.number BETWEEN 1 AND LEN(s0.[Social_Security_Number])
                 AND SUBSTRING(s0.[Social_Security_Number], v.number, 1) LIKE '[0-9]'
               FOR XML PATH('')), 1, 0, '') AS ssn_key
    FROM [Ankit_STS].[s_checks].[SSN_Based] s0
    WHERE NULLIF(LTRIM(RTRIM(s0.[Social_Security_Number])), '') IS NOT NULL
